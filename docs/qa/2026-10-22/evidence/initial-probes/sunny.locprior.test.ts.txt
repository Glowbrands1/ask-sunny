async function converse(role: string, scope: AccessScope, turns: string[]) {
  state.role = role;
  state.scope = scope;
  const messages: ChatMessage[] = [];
  const out: Awaited<ReturnType<typeof answerQuestion>>[] = [];
  let n = 0;
  for (const text of turns) {
    n += 1;
    const user = { id: `u${n}`, role: "user", content: text, createdAt: "2026-10-21T18:00:00Z" } as ChatMessage;
    const r = await answerQuestion({ question: text, mode: "standard", history: messages, questionMessageId: user.id, scopeId: "sun-tan-city", continueProposalTemplateKey: continuationFor(messages)?.templateKey, context: { userName: "Probe", locationName: "", todayIso: "2026-10-21" } } as never, { role: role as never, scope });
    out.push(r);
    messages.push(user, { id: `a${n}`, role: "assistant", content: String(r.content ?? ""), createdAt: "2026-10-21T18:00:01Z", ...(r.formProposal ? { formProposal: r.formProposal } : {}), ...(r.formSelection ? { formSelection: r.formSelection } : {}) } as ChatMessage);
  }
  return out;
}

/* --------------------------------------- criterion 3: prior coaching phrasing */
const PRIOR_PHRASES = [
  "I coached her on 10/1 for being late",
  "she was coached last week about lateness",
  "we had a talk with her about tardiness on 10/1",
  "I already spoke to her about being late",
  "she had a coaching on 10/1",
  "previous coaching 10/1 for lateness",
  "this is after her coaching on 10/1",
  "she's been coached before for this",
  "we discussed attendance with her on 10/1",
  "she got a verbal warning on 10/1",
  "I talked to pat about being late last month",
  "she was written up for this in september",
];
for (const phrase of PRIOR_PHRASES) {
  it(`3-ca ${phrase}`, { timeout: 60_000 }, async () => {
    let r: Awaited<ReturnType<typeof answerQuestion>> | undefined;
    let error: string | null = null;
    try {
      [r] = await converse("salon_director", { level: "salon", primaryAreaId: "loc-0310", alsoCoversAreaIds: [] }, [`create a verbal ca for pat rivera, she was late again today. ${phrase}`]);
    } catch (e) {
      error = String(e);
    }
    const content = String(r?.content ?? "");
    const notRecognised = /won't guess at it|any prior coaching or corrective action\?/i.test(content);
    const formDate = (r?.formProposal as { formDate?: string | null } | undefined)?.formDate ?? null;
    results.push({ criterion: 3, kind: "CA request", id: phrase, expected: "prior step recognised as history (not 'I won't guess'); the prior date is not the form date", actual: error ?? { recognised: !notRecognised, formDate, excerpt: content.replace(/\n+/g, " / ").slice(0, 500) }, pass: !error && !notRecognised && formDate !== "2026-10-01" });
  });
  it(`3-offer ${phrase}`, { timeout: 60_000 }, async () => {
    let r: Awaited<ReturnType<typeof answerQuestion>> | undefined;
    let error: string | null = null;
    try {
      [r] = await converse("salon_director", { level: "salon", primaryAreaId: "loc-0310", alsoCoversAreaIds: [] }, [`Pat Rivera was late again today. ${phrase}. what should I do?`]);
    } catch (e) {
      error = String(e);
    }
    const offered = JSON.stringify(r?.formSelection ?? null);
    results.push({ criterion: 3, kind: "suggestion", id: phrase, expected: "Corrective Action offered beside Coaching (prior step + recurrence)", actual: error ?? { offered: r?.formSelection ? offered.slice(0, 200) : "no suggestion" }, pass: !error && offered.includes("dpoa") });
  });
}
