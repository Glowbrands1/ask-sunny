/** QA PROBE (not committed): findings A/B/C through `proposeFormForTurn` directly, as the HR e2e harness drives it. */
import { writeFileSync } from "node:fs";
import { afterAll, it, vi } from "vitest";
import { fakeSupabase, type FakeStore } from "@/test/fake-supabase";
import type { AccessScope, ChatMessage } from "@/types";

const store: FakeStore = { form_instances: [], form_instance_values: [], form_instance_events: [], form_template_versions: [], form_templates: [], form_template_current: [], form_template_assets: [] };
const state = vi.hoisted(() => ({ directory: [] as unknown[] }));
vi.mock("@/lib/supabase/server", () => ({ getSupabaseAdmin: () => fakeSupabase(store) }));
vi.mock("@/lib/forms/employee-roster", async (importOriginal) => {
  const actual = await importOriginal<typeof import("@/lib/forms/employee-roster")>();
  return { ...actual, loadScopedRoster: async (scope: AccessScope | null) => actual.scopeRoster(state.directory as never, scope) };
});
process.env.NEXT_PUBLIC_DEMO_MODE = "false";
const { ensureTemplateLibrary, listTemplateSummaries } = await import("@/lib/forms/repository");
const { proposeFormForTurn } = await import("@/lib/ai/form-proposal");

const out: unknown[] = [];
const scope: AccessScope = { level: "salon", primaryAreaId: "loc-0310", alsoCoversAreaIds: [] };

async function play(turns: string[], keepOpen: boolean, directory: unknown[]) {
  state.directory = directory;
  const messages: ChatMessage[] = [];
  const rows: unknown[] = [];
  let open: string | undefined;
  let n = 0;
  for (const text of turns) {
    n += 1;
    const r = await proposeFormForTurn({
      history: messages, question: text, questionMessageId: `u${n}`,
      actor: { role: "salon_director" as never, scope }, summaries: await listTemplateSummaries(), today: "2026-10-21",
      policyManual: async () => null,
      ...(keepOpen && open ? { continueTemplateKey: open } : {}),
    });
    rows.push(r === null ? "NULL (turn left the forms path)" : { content: r.content.slice(0, 300), formProposal: r.formProposal ? { templateKey: r.formProposal.templateKey, employeeName: r.formProposal.employeeName, status: r.formProposal.status, formDate: r.formProposal.formDate, payrollDeduct: r.formProposal.payrollDeduct, warningLevel: r.formProposal.warningLevel } : null, formSelection: r.formSelection ? "picker" : null });
    if (r?.formProposal) open = r.formProposal.templateKey;
    messages.push({ id: `u${n}`, role: "user", content: text, createdAt: "2026-10-21T18:00:00Z" } as ChatMessage,
      { id: `a${n}`, role: "assistant", content: r?.content ?? "x", createdAt: "2026-10-21T18:00:01Z", ...(r?.formProposal ? { formProposal: r.formProposal } : {}) } as ChatMessage);
  }
  return rows;
}

const PAT = { id: "e-1", firstName: "Pat", lastName: "Rivera", preferredFirstName: null, employmentStatus: "active", salonIds: ["loc-0310"] };
const CASES: [string, string[], boolean, unknown[]][] = [
  ["A dir", ["my employee pat rivera was late again today, i need a coaching form"], true, [PAT]],
  ["A nodir", ["my employee pat rivera was late again today, i need a coaching form"], true, []],
  ["B open-key", ["create coaching form for pat rivera, late 3 times this week", "i already told you"], true, [PAT]],
  ["B no-key", ["create coaching form for pat rivera, late 3 times this week", "i already told you"], false, [PAT]],
  ["C", ["create verbal ca for pat rivera. she was late on 10/5 and 10/15. no payroll deduct"], true, [PAT]],
];
for (const [name, turns, keep, dir] of CASES) {
  it(name, { timeout: 60_000 }, async () => {
    for (const key of Object.keys(store) as (keyof FakeStore)[]) store[key] = [];
    vi.useFakeTimers({ toFake: ["Date"] }); vi.setSystemTime(new Date("2026-10-21T18:00:00Z"));
    await ensureTemplateLibrary("system");
    let rows: unknown;
    try { rows = await play(turns, keep, dir); } catch (e) { rows = `THREW ${String(e)}`; }
    out.push({ name, turns, rows });
  });
}
afterAll(() => writeFileSync(process.env.PROBE_OUT!, JSON.stringify(out, null, 2)));
