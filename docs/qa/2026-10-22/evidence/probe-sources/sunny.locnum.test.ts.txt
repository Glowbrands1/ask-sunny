function openAfter(responses: Response[]): string | null {
  const messages: ChatMessage[] = [];
  responses.forEach((r, i) =>
    messages.push(
      { id: `u${i}`, role: "user", content: "x", createdAt: "2026-10-21T18:00:00Z" } as ChatMessage,
      { id: `a${i}`, role: "assistant", content: String(r.content), createdAt: "2026-10-21T18:00:01Z", ...(r.formProposal ? { formProposal: r.formProposal } : {}), ...(r.formSelection ? { formSelection: r.formSelection } : {}) } as ChatMessage,
    ),
  );
  return continuationFor(messages)?.templateKey ?? null;
}

const results: unknown[] = [];

beforeEach(async () => {
  for (const key of Object.keys(store) as (keyof FakeStore)[]) store[key] = [];
  vi.useFakeTimers({ toFake: ["Date"] });
  vi.setSystemTime(new Date("2026-10-21T18:00:00Z"));
  await ensureTemplateLibrary("system");
});

for (const scenario of SCENARIOS) {
  it(`${scenario.id} ${scenario.group}`, { timeout: 60_000 }, async () => {
    state.directory = scenario.directory ?? [];
    const actor = scenario.actor ?? SD;
    const messages: ChatMessage[] = [];
    const responses: Response[] = [];
    let error: string | null = null;
    try {
      let n = 0;
      for (const text of scenario.turns) {
        n += 1;
        const user = { id: `${scenario.id}-u${n}`, role: "user", content: text, createdAt: "2026-10-21T18:00:00Z" } as ChatMessage;
        const response = await answerQuestion(
          {
            question: text,
            mode: "standard",
            history: messages,
            questionMessageId: user.id,
            scopeId: "sun-tan-city",
            continueProposalTemplateKey: continuationFor(messages)?.templateKey,
            context: { userName: "Probe Manager", locationName: "NE Lincoln 27th Street", todayIso: "2026-10-21" },
          } as never,
          { role: actor.role as never, scope: actor.scope },
        );
        responses.push(response);
        messages.push(user, {
          id: `${scenario.id}-a${n}`,
          role: "assistant",
          content: String(response.content ?? ""),
          createdAt: "2026-10-21T18:00:01Z",
          ...(response.formProposal ? { formProposal: response.formProposal } : {}),
          ...(response.formSelection ? { formSelection: response.formSelection } : {}),
        } as ChatMessage);
      }
    } catch (caught) {
      error = caught instanceof Error ? `${caught.name}: ${caught.message}` : String(caught);
    }
    let verdict: [boolean, string] = [false, "not evaluated"];
    if (!error) {
      try {
        verdict = scenario.check(responses);
      } catch (caught) {
        verdict = [false, `check threw: ${String(caught)}`];
      }
    }
    results.push({
      id: scenario.id,
      group: scenario.group,
      actor: actor.role,
      turns: scenario.turns,
      expected: scenario.expected,
      pass: !error && verdict[0],
      actual: error ?? verdict[1],
      raw: responses.map((r) => ({ content: r.content, formProposal: r.formProposal ?? null, formSelection: r.formSelection ?? null, coverage: r.coverage ?? null })),
    });
  });
}

afterAll(() => {
  writeFileSync(process.env.PROBE_OUT ?? "probe-out.json", JSON.stringify(results, null, 2));
});
