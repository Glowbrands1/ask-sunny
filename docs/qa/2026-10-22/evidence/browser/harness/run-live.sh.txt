#!/usr/bin/env bash
# QA (not committed): integration + browser validation for ONE app, on the disposable local stack.
# usage: run-live.sh <app-dir> <sunny|bubbles> <evidence-dir>
set -uo pipefail
APP_DIR=$1; APP=$2; OUT=$3
L=$(cd "$(dirname "$0")" && pwd)
STACK_DIR=/tmp/qa-stack-$APP
mkdir -p "$OUT/shots"
export LOCAL_STACK_DIR=$STACK_DIR POSTGREST_BIN=/tmp/postgrest
cd "$APP_DIR"
# Refuse to run against a server or proxy left over from an earlier run (a stale server once answered for a new one).
if curl -s -o /dev/null http://127.0.0.1:3200/ || curl -s -o /dev/null http://127.0.0.1:54331/; then echo "ABORT: port 3200 or 54331 already in use" | tee -a "$OUT/$APP-steps.txt"; exit 2; fi
echo "== [$APP] stack up"; npm run -s stack:up > "$OUT/$APP-stack-up.log" 2>&1; echo "stack:up exit $?" | tee -a "$OUT/$APP-steps.txt"
echo "== [$APP] local-stack integration tests"; npm run -s test:local-stack > "$OUT/$APP-local-stack.log" 2>&1; echo "test:local-stack exit $?" | tee -a "$OUT/$APP-steps.txt"
grep -E "Test Files|^\s+Tests " "$OUT/$APP-local-stack.log" | tee -a "$OUT/$APP-steps.txt"
echo "== [$APP] seed accounts"; node "$L/seed-users.mjs" "$APP_DIR" "$STACK_DIR/env" "$APP" > "$OUT/$APP-seed.log" 2>&1; echo "seed exit $?" | tee -a "$OUT/$APP-steps.txt"
set -a; . "$STACK_DIR/env"; set +a
# QA-only CORS proxy: the local gateway sends no CORS headers (the real Supabase gateway does).
node "$L/cors-proxy.mjs" 54331 54321 http://127.0.0.1:3200 > "$OUT/$APP-cors-proxy.log" 2>&1 & echo $! > "$OUT/$APP-cors.pid"
sleep 1
export NEXT_PUBLIC_DEMO_MODE=false NEXT_PUBLIC_SUPABASE_URL="http://127.0.0.1:54331" NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY="$LOCAL_ANON_KEY" SUPABASE_SECRET_KEY="$LOCAL_SERVICE_ROLE_KEY" ANTHROPIC_API_KEY=sk-ant-qa-local-placeholder FORMS_TEMPLATE_SYNC_ENABLED=true NEXT_PUBLIC_SITE_URL=http://127.0.0.1:3200
echo "== [$APP] next build (live mode, local stack)"; npx next build > "$OUT/$APP-build.log" 2>&1; echo "next build exit $?" | tee -a "$OUT/$APP-steps.txt"
echo "== [$APP] next start"; npx next start -p 3200 > "$OUT/$APP-server.log" 2>&1 & echo $! > "$OUT/$APP-server.pid"; for i in $(seq 1 60); do curl -fsS -o /dev/null http://127.0.0.1:3200/login && break; sleep 2; done; if grep -q EADDRINUSE "$OUT/$APP-server.log"; then echo "ABORT: this run's server could not bind 3200" | tee -a "$OUT/$APP-steps.txt"; exit 3; fi; grep -m1 "Ready in" "$OUT/$APP-server.log" | tee -a "$OUT/$APP-steps.txt"; curl -s -o /dev/null -w "login page HTTP %{http_code}\n" http://127.0.0.1:3200/login | tee -a "$OUT/$APP-steps.txt"
echo "== [$APP] playwright"; rm -f "$OUT/$APP-browser.jsonl"
(cd "$L" && QA_APP="$(cat "$L/$APP.app.json")" QA_OUT="$OUT/$APP-browser.jsonl" QA_SHOTS="$OUT/shots" QA_BASE_URL=http://127.0.0.1:3200 PLAYWRIGHT_BROWSERS_PATH=/opt/pw-browsers npx playwright test -c pw.config.ts > "$OUT/$APP-playwright.log" 2>&1); echo "playwright exit $?" | tee -a "$OUT/$APP-steps.txt"
tail -15 "$OUT/$APP-playwright.log" | tee -a "$OUT/$APP-steps.txt"
echo "== [$APP] teardown"; kill $(cat "$OUT/$APP-server.pid") 2>/dev/null; kill $(cat "$OUT/$APP-cors.pid") 2>/dev/null; for p in $(ps -eo pid,args | awk '$2 ~ /^next-server/ {print $1}'); do kill $p 2>/dev/null; done; sleep 2; npm run -s stack:down > "$OUT/$APP-stack-down.log" 2>&1; echo "stack:down exit $?" | tee -a "$OUT/$APP-steps.txt"
