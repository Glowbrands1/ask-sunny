// QA ONLY (not committed): the disposable local stack's gateway sends no CORS
// headers, which the real Supabase gateway does. This adds them for the one
// test origin and forwards everything else unchanged. 127.0.0.1 only.
import http from "node:http";
const [listen, target, origin] = [Number(process.argv[2]), Number(process.argv[3]), process.argv[4]];
const cors = {
  "access-control-allow-origin": origin,
  "access-control-allow-credentials": "true",
  "access-control-allow-methods": "GET,POST,PUT,PATCH,DELETE,OPTIONS",
  "access-control-allow-headers": "authorization,apikey,content-type,x-client-info,x-supabase-api-version,prefer,range,accept-profile,content-profile",
  "access-control-expose-headers": "content-range,x-supabase-api-version",
};
http.createServer((req, res) => {
  if (req.method === "OPTIONS") { res.writeHead(204, cors).end(); return; }
  const up = http.request({ host: "127.0.0.1", port: target, method: req.method, path: req.url, headers: req.headers }, (reply) => {
    const headers = { ...reply.headers };
    for (const k of Object.keys(headers)) if (k.startsWith("access-control-")) delete headers[k];
    res.writeHead(reply.statusCode ?? 502, { ...headers, ...cors });
    reply.pipe(res);
  });
  up.on("error", () => res.writeHead(502, cors).end("upstream down"));
  req.pipe(up);
}).listen(listen, "127.0.0.1", () => console.log(`cors proxy ${listen} -> ${target} for ${origin}`));
