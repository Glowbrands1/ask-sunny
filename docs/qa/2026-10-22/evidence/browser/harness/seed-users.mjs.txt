// Usage: node seed-users.mjs <app-dir> <stack-env-file> <app: sunny|bubbles>
import { readFileSync } from "node:fs";
import { createRequire } from "node:module";
const [appDir, envFile, app] = process.argv.slice(2);
const require = createRequire(`${appDir}/package.json`);
const { createClient } = require("@supabase/supabase-js");
const env = Object.fromEntries(readFileSync(envFile, "utf8").split("\n").filter((l) => l.includes("=")).map((l) => [l.slice(0, l.indexOf("=")), l.slice(l.indexOf("=") + 1)]));
const url = env.LOCAL_STACK_URL;
if (new URL(url).hostname !== "127.0.0.1") throw new Error("refusing: not the local stack");
const key = env.LOCAL_SERVICE_ROLE_KEY;
const admin = createClient(url, key, { auth: { persistSession: false, autoRefreshToken: false } });
const PASSWORD = "<local-only QA password>";
// (Sunny accounts only in this copy)
const users = [
      ["qa-admin@local.test", "admin", { scope_level: "global", scope_primary_area_id: null, scope_also_covers_area_ids: [] }],
      ["qa-sd@local.test", "salon_director", { scope_level: "salon", scope_primary_area_id: "loc-0310", scope_also_covers_area_ids: ["loc-0311"] }],
    ];
for (const [email, role, scope] of users) {
  const { data, error } = await admin.auth.admin.createUser({ email, password: PASSWORD, email_confirm: true });
  if (error || !data.user) throw new Error(`createUser ${email}: ${error?.message}`);
  const { error: e2 } = await admin.from("app_users").insert({ id: data.user.id, email, display_name: email.split("@")[0], role, status: "active", ...scope });
  if (e2) throw new Error(`app_users ${email}: ${e2.message}`);
  console.log("created", email, role, JSON.stringify(scope));
}
