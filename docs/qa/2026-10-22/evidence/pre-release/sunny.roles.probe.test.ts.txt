const results: Record<string, unknown>[] = [];
const PAT = { id: "e-1", firstName: "Pat", lastName: "Rivera", preferredFirstName: null, employmentStatus: "active", salonIds: ["loc-0310"] };
beforeEach(async () => {
  for (const key of Object.keys(store) as (keyof FakeStore)[]) store[key] = [];
  vi.useFakeTimers({ toFake: ["Date"] });
  vi.setSystemTime(new Date("2026-10-21T18:00:00Z"));
  await ensureTemplateLibrary("system");
  state.directory = [PAT];
});
const SALON: AccessScope = { level: "salon", primaryAreaId: "loc-0310", alsoCoversAreaIds: [] };
const permissions = await import("@/lib/permissions");
it("matrix", () => {
  for (const role of ["employee", "assistant_salon_director", "salon_director"]) {
    results.push({ kind: "matrix", role, permissions: [...(permissions.DEFAULT_PERMISSION_MATRIX as Record<string, string[]>)[role]!].sort() });
  }
});
for (const role of ["employee", "assistant_salon_director", "salon_director"]) {
  it(`create every template as ${role}`, { timeout: 120_000 }, async () => {
    state.role = role; state.scope = SALON;
    const keys = ((store.form_templates ?? []) as { key: string; required_permission: string; active: boolean }[]).map((t) => ({ key: t.key, perm: t.required_permission, active: t.active }));
    for (const t of keys) {
      let status: number | string;
      try {
        const response = await instancesRoute.POST(new Request("https://app.test/api/forms/instances", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ templateKey: t.key, employeeName: "Pat Rivera", locationId: "loc-0310", source: "manual" }) }));
        status = response.status;
      } catch (e) { status = `threw ${String(e)}`; }
      results.push({ kind: "create", role, template: t.key, requiredPermission: t.perm, active: t.active, status });
    }
  });
  it(`chat card as ${role}`, { timeout: 60_000 }, async () => {
    state.role = role; state.scope = SALON;
    const q = "create a coaching form for pat rivera, late 3 times this week";
    let out: unknown;
    try {
      const a = (await answerQuestion({ question: q, mode: "standard", history: [] as ChatMessage[], questionMessageId: "u1", scopeId: "sun-tan-city", context: { userName: "Probe", locationName: "", todayIso: "2026-10-21" } } as never, { role: role as never, scope: SALON })) as Record<string, unknown>;
      out = { proposal: Boolean(a.formProposal), selection: Boolean(a.formSelection), content: String(a.content ?? "").slice(0, 160) };
    } catch (e) { out = `threw ${String(e)}`; }
    results.push({ kind: "chat", role, question: q, out });
  });
}
afterAll(() => writeFileSync(process.env.QA_OUT!, JSON.stringify(results, null, 1)));
