import { it } from "vitest";
import { statedWarningLevel } from "@/lib/forms/warning-level";
it("probe", () => {
  process.env.LEVEL_AFTER_FORM_NAME_ENABLED = "true";
  const out: string[] = [];
  for (const t of ["she had a ca verbal on 9/1 for the same thing", "her last ca written was in august", "she had a verbal warning on 9/1 for the same thing"]) out.push(`${t} => ${statedWarningLevel(t, "2026-11-10")}`);
  require("node:fs").writeFileSync(process.env.PROBE_OUT!, out.join("\n"));
});
